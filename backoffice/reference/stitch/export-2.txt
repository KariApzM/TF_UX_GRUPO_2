  function showToast(message, icon = 'check_circle') {
    const toast = document.getElementById('toast-action');
    const msg = document.getElementById('toast-msg');
    const icn = document.getElementById('toast-icon');
    if (!toast || !msg || !icn) return;
    
    msg.textContent = message;
    icn.textContent = icon;
    toast.classList.remove('translate-y-32', 'opacity-0', 'pointer-events-none');
    toast.classList.add('translate-y-0', 'opacity-100');

    setTimeout(() => {
      closeToast();
    }, 3800);
  }

  function closeToast() {
    const toast = document.getElementById('toast-action');
    if (!toast) return;
    toast.classList.add('translate-y-32', 'opacity-0', 'pointer-events-none');
    toast.classList.remove('translate-y-0', 'opacity-100');
  }

  const btnSync = document.getElementById('btn-sync-gps');
  if (btnSync) {
    btnSync.addEventListener('click', () => {
      const icon = document.getElementById('sync-icon');
      const timeLabel = document.getElementById('last-sync-time');
      if (icon) icon.classList.add('animate-spin');
      btnSync.disabled = true;

      setTimeout(() => {
        if (icon) icon.classList.remove('animate-spin');
        btnSync.disabled = false;
        if (timeLabel) timeLabel.textContent = 'Actualizado hace 1s';
        showToast('Coordenadas GPS sincronizadas con la torre de control.', 'satellite_alt');
      }, 900);
    });
  }

  const btnNotify = document.getElementById('btn-notify-store');
  if (btnNotify) {
    btnNotify.addEventListener('click', () => {
      showToast('Alerta enviada: El equipo de tienda #412 prepara la rampa de descarga.', 'campaign');
    });
  }

  const btnIncident = document.getElementById('btn-report-incident');
  if (btnIncident) {
    btnIncident.addEventListener('click', () => {
      showToast('Abriendo canal prioritario con la central de monitoreo...', 'security');
    });
  }
